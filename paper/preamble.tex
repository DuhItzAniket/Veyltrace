% ---------------------------------------------------------------- generated numbers
% \resnum{<experiment_id>}{<selector>} prints a result number once the generator
% defines it with \defresnum{<experiment_id>}{<selector>}{<value>} in
% generated/numbers.tex; until then it prints a red blank naming its source.
% No number in the paper is typed by hand (docs/11 §10, AGENTS.md §4.8).
\makeatletter
\newcommand{\defresnum}[3]{\expandafter\def\csname vt@num@\detokenize{#1|#2}\endcsname{#3}}
\newcommand{\resnum}[2]{%
  \ifcsname vt@num@\detokenize{#1|#2}\endcsname
    \csname vt@num@\detokenize{#1|#2}\endcsname
  \else
    \textcolor{red}{[\,\textsf{\detokenize{#1}}: \detokenize{#2}\,]}%
  \fi}
\makeatother
\InputIfFileExists{generated/numbers}{}{}

% ---------------------------------------------------------------- generated tables and figures
% \resulttable[*]{<label>}{<caption>}{<experiment ids>}{<name>} inputs
% generated/<name>.tex (R-17 writes bench/results/tables/<name>.md and the .tex copy);
% until then it shows a placeholder that names the experiments and the file.
\makeatletter
\newcommand{\vt@placeholder}[3]{%
  \fbox{\parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{\small
    \textbf{Placeholder (#1).} Source experiments: \texttt{\detokenize{#2}}.
    Generated by R-17 (\#46) as \texttt{bench/results/tables/\detokenize{#3}.md},
    copied to \texttt{paper/generated/\detokenize{#3}}.}}}
\NewDocumentCommand{\resulttable}{s m m m m}{%
  \IfBooleanTF{#1}{\begin{table*}[t]}{\begin{table}[t]}
  \caption{#3}\label{#2}
  \IfFileExists{generated/#5.tex}{\input{generated/#5.tex}}{\vt@placeholder{table}{#4}{#5}}
  \IfBooleanTF{#1}{\end{table*}}{\end{table}}}
\NewDocumentCommand{\resultfigure}{s m m m m}{%
  \IfBooleanTF{#1}{\begin{figure*}[t]}{\begin{figure}[t]}
  \centering
  \IfFileExists{generated/#5.pdf}{\includegraphics[width=\linewidth]{generated/#5.pdf}}{\vt@placeholder{figure}{#4}{#5}}
  \caption{#3}\label{#2}
  \IfBooleanTF{#1}{\end{figure*}}{\end{figure}}}
% \diagram[*]{<label>}{<caption>}{<name>}{<what to draw>}: hand-drawn figures live in
% figures/<name>.pdf (source next to it); until then a box says what to draw.
\NewDocumentCommand{\diagram}{s m m m m}{%
  \IfBooleanTF{#1}{\begin{figure*}[t]}{\begin{figure}[t]}
  \centering
  \IfFileExists{figures/#4.pdf}{\includegraphics[width=\linewidth]{figures/#4.pdf}}{%
    \fbox{\parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{\small
      \textbf{Diagram to draw} (\texttt{paper/figures/\detokenize{#4}.pdf}): #5}}}
  \caption{#3}\label{#2}
  \IfBooleanTF{#1}{\end{figure*}}{\end{figure}}}
\makeatother
