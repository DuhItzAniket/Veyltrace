% ---------------------------------------------------------------- generated tables and figures
% \resulttable[*]{<label>}{<caption>}{<experiment ids>}{<name>} inputs generated/<name>-paper.tex
% (R-17's compact paper view: headline columns, chosen rows) when it exists, else
% generated/<name>.tex (copies of R-17's bench/results/tables/), inside the paper's own float: R-17's
% table float, caption and label are unwrapped, its notes and source lines kept, and its
% caption printed below them. \resultfigure inputs generated/<name>.tex (R-17's SVG panels
% as TikZ, written by scripts/paper_generated.py), else generated/<name>.pdf. Until a file
% exists, a placeholder names the experiments and R-17's output.
\makeatletter
\newcommand{\vt@placeholder}[3]{% kind, experiment ids, where R-17 writes it (sans serif, so long ids break at hyphens)
  \fbox{\parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{\small\sloppy
    \textbf{Placeholder (#1).} Source experiments: \textsf{\detokenize{#2}}.
    Generated by R-17 (\#46): #3.}}}
\newcommand{\vt@gencap}{}
\newcommand{\vt@inputgenerated}[1]{%
  \begingroup
  \renewenvironment{table}[1][]{}{}%
  \renewcommand{\caption}[2][]{\gdef\vt@gencap{##2}}%
  \renewcommand{\label}[1]{}%
  \input{#1}%
  \par{\footnotesize Generated caption: \vt@gencap\par}%
  \endgroup}
% Generated text names other generated tables and figures with \vtref{<name>}, \vtfullref{<name>}
% and \vtfigref{<name>} (R-17 replaces the generator's own numbers, "Table 7c", which differ from
% LaTeX's). \resulttable places \label{vtgen:<name>}, \resulttablefull and \resulttableentry
% \label{vtgen:<name>-full}, \resultfigure \label{vtgen:<name>}. \vtref prints the main-text table,
% else its appendix grid; \vtfullref the appendix grid. A target the paper does not print never
% reads "??": the reference names its released file instead and LaTeX warns "vtref".
\newcommand{\vt@reffile}[3]{% kind, name, released file
  \PackageWarning{vtref}{#1 #2 is not printed; naming #3}\textsf{\detokenize{#3}}}
\DeclareRobustCommand{\vtref}[1]{%
  \ifcsname r@vtgen:#1\endcsname Table~\ref{vtgen:#1}%
  \else\ifcsname r@vtgen:#1-full\endcsname Table~\ref{vtgen:#1-full}%
  \else\vt@reffile{table}{#1}{bench/results/tables/#1.md}\fi\fi}
\DeclareRobustCommand{\vtfullref}[1]{%
  \ifcsname r@vtgen:#1-full\endcsname Table~\ref{vtgen:#1-full}%
  \else\vt@reffile{full grid}{#1}{bench/results/tables/#1.md}\fi}
\DeclareRobustCommand{\vtfigref}[1]{%
  \ifcsname r@vtgen:#1\endcsname Figure~\ref{vtgen:#1}%
  \else\vt@reffile{figure}{#1}{bench/results/figures/#1.csv}\fi}
\NewDocumentCommand{\resulttable}{s m m m m}{%
  \IfBooleanTF{#1}{\begin{table*}[t]}{\begin{table}[t]}
  \caption{#3}\label{#2}\label{vtgen:#5}
  \IfFileExists{generated/#5-paper.tex}{\vt@inputgenerated{generated/#5-paper.tex}}{%
    \IfFileExists{generated/#5.tex}{\vt@inputgenerated{generated/#5.tex}}{\vt@placeholder{table}{#4}{\textsf{bench/results/tables/\detokenize{#5}.\{md,tex\}}, copied to \textsf{paper/generated/\detokenize{#5}.tex}}}}
  \IfBooleanTF{#1}{\end{table*}}{\end{table}}}
% \resulttablefull{<label>}{<caption>}{<name>} prints R-17's full grid generated/<name>-full.tex
% (one longtable per column part, breaking across pages) for the appendix. longtable needs one
% column, so the first full grid switches the rest of the document to \onecolumn (a new page).
% The generated file calls \vtfullcaption (part 1: caption and label), \vtfullnext{<k>} and
% \vtfullpart{<k>}{<n>} (later parts) and \vtfullcont{<k>}{<n>} (the head of a continued page).
% Its first line, \vtfullsize{<rows>}{<columns>}, gives the grid's size (as in the .md), which
% \resulttableentry reads alone.
\newcommand{\vt@fullcaption}{}
\newcommand{\vt@fulllabel}{}
\newcommand{\vt@fullname}{}
\newcommand{\vtfullsize}[2]{}
\newcommand{\vtfullcaption}{\caption{\vt@fullcaption}\label{\vt@fulllabel}\label{vtgen:\vt@fullname-full}}
% Each longtable steps the table counter, so \vtfullnext{<k>}, before a later part, keeps its
% table's number and gives hyperref a distinct anchor.
\newcommand{\vtfullnext}[1]{\addtocounter{table}{-1}\def\theHtable{\arabic{table}.#1}}
\newcommand{\vtfullpart}[2]{\caption*{Table~\ref{\vt@fulllabel}, continued: columns, part #1 of #2}}
\newcommand{\vtfullcont}[2]{\textit{Table~\ref{\vt@fulllabel}, continued} (columns, part #1 of #2)}
\newcommand{\resulttablefull}[3]{%
  \if@twocolumn\onecolumn\fi
  \IfFileExists{generated/#3-full.tex}{%
    \begingroup\def\vt@fullcaption{#2}\def\vt@fulllabel{#1}\def\vt@fullname{#3}%
    \input{generated/#3-full.tex}\endgroup\par\addvspace{\bigskipamount}}{%
    \vt@placeholder{full table}{#3}{\textsf{bench/results/tables/\detokenize{#3}-full.tex}, copied to \textsf{paper/generated/}}\par\addvspace{\bigskipamount}}}
% \resulttableentry{<label>}{<caption>}{<name>} stands in the appendix for a full grid too long to
% print (lead decision, 27 Nov: the appendix prints grids of about four pages at most): a one-row
% table naming the released files and the grid's size, read from \vtfullsize in
% generated/<name>-full.tex (\endinput stops reading the file there).
\newcommand{\vt@rows}{}
\newcommand{\vt@cols}{}
\newcommand{\vt@readsize}[2]{\gdef\vt@rows{#1}\gdef\vt@cols{#2}\endinput}
\newcommand{\resulttableentry}[3]{%
  \if@twocolumn\onecolumn\fi
  \IfFileExists{generated/#3-full.tex}{%
    \begingroup
    \let\vtfullsize\vt@readsize
    \input{generated/#3-full.tex}\endgroup
    \begin{longtable}{@{}>{\raggedright\arraybackslash}p{\linewidth}@{}}
    \caption{#2}\label{#1}\label{vtgen:#3-full} \\
    \hline
    Full grid, too long to print here: \textsf{bench/results/tables/\detokenize{#3}.md}
    (\vt@rows\ rows $\times$ \vt@cols\ columns, the source column included; also as LaTeX in
    \textsf{\detokenize{#3}.tex} and, as longtables, \textsf{\detokenize{#3}-full.tex}). \\
    \hline
    \end{longtable}\par\addvspace{\bigskipamount}}{%
    \vt@placeholder{full table}{#3}{\textsf{bench/results/tables/\detokenize{#3}-full.tex}, copied to \textsf{paper/generated/}}\par\addvspace{\bigskipamount}}}
% A generated figure file is \vtfigpanels{<n>} followed by one \vtfigpanel{<tikzpicture>} per
% SVG panel, drawn at 1 px = 1 pt and scaled as a whole (text included, as an SVG viewer
% scales it). Panels sit \vt@perrow to a row: 2 in a starred figure, 1 (stacked) otherwise,
% so a three-panel figure* takes two rows and fits the text height.
\newcount\vt@perrow
\newcount\vt@inrow
\newlength{\vt@panelw}
\newcommand{\vtfigpanels}[1]{%
  \vt@inrow=0
  \ifnum#1<\vt@perrow \vt@perrow=#1\relax\fi
  \setlength{\vt@panelw}{\dimexpr(\linewidth-1em*\numexpr\vt@perrow-1\relax)/\vt@perrow\relax}}
\newcommand{\vtfigpanel}[1]{%
  \ifnum\vt@inrow=\vt@perrow \par\medskip\vt@inrow=0 \fi
  \ifnum\vt@inrow>0 \hfill\fi
  \advance\vt@inrow by 1
  \resizebox{\vt@panelw}{!}{#1}}
\NewDocumentCommand{\resultfigure}{s m m m m}{%
  \IfBooleanTF{#1}{\begin{figure*}[t]\vt@perrow=2 }{\begin{figure}[t]\vt@perrow=1 }
  \centering
  \IfFileExists{generated/#5.tex}{\input{generated/#5.tex}\par}{%
    \IfFileExists{generated/#5.pdf}{\includegraphics[width=\linewidth]{generated/#5.pdf}}{\vt@placeholder{figure}{#4}{\textsf{bench/results/figures/\detokenize{#5}.csv} and \textsf{\detokenize{#5}[-\textless panel\textgreater].svg}, drawn as TikZ in \textsf{paper/generated/\detokenize{#5}.tex} by \textsf{scripts/paper\_generated.py}}}}
  \caption{#3}\label{#2}\label{vtgen:#5}
  \IfBooleanTF{#1}{\end{figure*}}{\end{figure}}}
% \diagram[*]{<label>}{<caption>}{<name>}{<what to draw>}: hand-drawn figures live in
% figures/<name>.pdf (source next to it); until then a box says what to draw.
\NewDocumentCommand{\diagram}{s m m m m}{%
  \IfBooleanTF{#1}{\begin{figure*}[t]}{\begin{figure}[t]}
  \centering
  \IfFileExists{figures/#4.pdf}{\includegraphics[width=\linewidth]{figures/#4.pdf}}{%
    \fbox{\parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{\small
      \textbf{Diagram to draw} (\texttt{paper/figures/\detokenize{#4}.pdf}): #5}}}
  \caption{#3}\label{#2}
  \IfBooleanTF{#1}{\end{figure*}}{\end{figure}}}
\makeatother
